cmake_minimum_required(VERSION 3.16)
project(touch_card_core LANGUAGES CXX)
enable_testing()
foreach(LANGUAGE_DEFAULT IN ITEMS 0 1)
  add_executable(language_power_${LANGUAGE_DEFAULT}_test firmware/test/language_power_test.cpp)
  target_include_directories(language_power_${LANGUAGE_DEFAULT}_test PRIVATE firmware/lib/CardCore/src)
  target_compile_features(language_power_${LANGUAGE_DEFAULT}_test PRIVATE cxx_std_17)
  target_compile_definitions(language_power_${LANGUAGE_DEFAULT}_test PRIVATE TOUCH_CARD_DEFAULT_LANGUAGE=${LANGUAGE_DEFAULT})
  target_compile_options(language_power_${LANGUAGE_DEFAULT}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined)
  target_link_options(language_power_${LANGUAGE_DEFAULT}_test PRIVATE -fsanitize=address,undefined)
  add_test(NAME language_power_${LANGUAGE_DEFAULT} COMMAND language_power_${LANGUAGE_DEFAULT}_test)
endforeach()
