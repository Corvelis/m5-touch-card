cmake_minimum_required(VERSION 3.16)
project(paper_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GFX "${ROOT}/firmware/.pio/libdeps/paper-mono/M5GFX/src" CACHE PATH "Pinned M5GFX 0.2.27 source directory")
file(GLOB_RECURSE GFX_SOURCES "${GFX}/*.cpp" "${GFX}/*.c")
add_library(gfx STATIC ${GFX_SOURCES})
target_include_directories(gfx PUBLIC "${GFX}")
target_link_libraries(gfx PUBLIC PkgConfig::SDL)
target_compile_options(gfx PRIVATE -w)
add_executable(paper_preview main.cpp "${ROOT}/firmware/lib/PaperUI/src/PaperFont.cpp")
target_include_directories(paper_preview PRIVATE "${ROOT}/firmware/lib/PaperUI/src"
                           "${ROOT}/firmware/lib/CardCore/src" "${ROOT}/firmware/src")
target_compile_definitions(paper_preview PRIVATE TOUCH_CARD_HOST=1)
target_link_libraries(paper_preview PRIVATE gfx)
add_executable(stack_digits_preview stack_digits.cpp)
target_include_directories(stack_digits_preview PRIVATE "${ROOT}/firmware/src"
                           "${ROOT}/firmware/lib/CardCore/src")
target_compile_options(stack_digits_preview PRIVATE -UNDEBUG)
target_link_libraries(stack_digits_preview PRIVATE gfx)
